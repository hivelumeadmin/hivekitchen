### AC1 — New `ChildAllergensRepository` with encrypted `declare`

Create `apps/api/src/modules/children/child-allergens.repository.ts`:

```typescript
import { Buffer } from 'node:buffer';
import type { SupabaseClient } from '@supabase/supabase-js';
import { encryptField, normalizedHash } from '../../lib/envelope-encryption.js';
import { getOrCreateHouseholdDek } from '../../lib/household-key.js';

export class ChildAllergensRepository {
  constructor(
    private readonly client: SupabaseClient,
    private readonly kek: Buffer | null,
  ) {}

  async declare(
    householdId: string,
    childId: string,
    allergen: string,
    source: string,
  ): Promise<{ child_allergen_id: string; was_existing: boolean }> {
    const dek = await getOrCreateHouseholdDek(this.client, this.kek, householdId);
    const ciphertext = encryptField(allergen, dek);
    const hash = normalizedHash(allergen);

    // Upsert using the (child_id, allergen_hash) unique index.
    // Supabase upsert with ignoreDuplicates:false does a no-change UPDATE on
    // conflict, which returns the existing row — gives us the id either way.
    const { data, error } = await this.client
      .from('child_allergens')
      .upsert(
        {
          household_id: householdId,
          child_id: childId,
          allergen: ciphertext,
          allergen_hash: hash,
          source,
          updated_at: new Date().toISOString(),
        },
        { onConflict: 'child_id,allergen_hash', ignoreDuplicates: false },
      )
      .select('id, created_at, updated_at')
      .maybeSingle();

    if (error !== null && error !== undefined) {
      throw new Error(`child_allergens.declare: ${error.message}`);
    }
    if (data === null) {
      throw new Error(`child_allergens.declare: no row returned for child ${childId}`);
    }
    const row = data as { id: string; created_at: string; updated_at: string };
    // was_existing: if created_at !== updated_at the row was updated (existed already).
    // On first insert both are set by the server to the same value; upsert sets
    // updated_at explicitly on conflict so the timestamps diverge.
    // More reliable: assume was_existing=false only when created_at ≈ updated_at
    // (within 1 second). This is a best-effort heuristic — no loss of correctness.
    const was_existing = Math.abs(
      new Date(row.updated_at).getTime() - new Date(row.created_at).getTime()
    ) > 1000;

    return { child_allergen_id: row.id, was_existing };
  }
}
```

### AC2 — New `DietaryPreferencesRepository` with `declare`

Create `apps/api/src/modules/dietary-preferences/dietary-preferences.repository.ts`:

```typescript
import type { SupabaseClient } from '@supabase/supabase-js';

export class DietaryPreferencesRepository {
  constructor(private readonly client: SupabaseClient) {}

  async declare(
    householdId: string,
    childId: string | null,
    tag: string,
    enforcement: string,
    source: string,
  ): Promise<{ dietary_id: string; was_existing: boolean }> {
    // Unique index: dietary_preferences_scope_tag_uniq ON
    //   (household_id, COALESCE(child_id, '00000000-0000-0000-0000-000000000000'::uuid), tag)
    // tag is a vocabulary key (closed vocab) — NOT encrypted.
    const { data, error } = await this.client
      .from('dietary_preferences')
      .upsert(
        {
          household_id: householdId,
          child_id: childId,
          tag,
          enforcement,
          source,
          updated_at: new Date().toISOString(),
        },
        {
          onConflict: 'household_id,child_id,tag',
          ignoreDuplicates: false,
        },
      )
      .select('id, created_at, updated_at')
      .maybeSingle();

    if (error !== null && error !== undefined) {
      throw new Error(`dietary_preferences.declare: ${error.message}`);
    }
    if (data === null) {
      throw new Error(`dietary_preferences.declare: no row returned`);
    }
    const row = data as { id: string; created_at: string; updated_at: string };
    const was_existing = Math.abs(
      new Date(row.updated_at).getTime() - new Date(row.created_at).getTime()
    ) > 1000;

    return { dietary_id: row.id, was_existing };
  }
}
```

**IMPORTANT — unique index limitation:** Supabase upsert `onConflict` parameter must match the index columns exactly. The `dietary_preferences_scope_tag_uniq` index uses `COALESCE(child_id, sentinel_uuid)` which is a functional expression index. Supabase may not support `onConflict` on functional indexes. If the upsert fails:
- Alternative: insert, catch Postgres 23505 error code, then SELECT existing row by `(household_id, child_id IS NULL or child_id = ?, tag)` to get the id.
- Use `isUniqueViolation(err)` from `../../common/errors.js` to detect 23505.

**AC6.1 — "No known allergens" mutual exclusion:**

The current chip tap handler in `OnboardingText.tsx` simply toggles chips in the `chipSelections` array. For M2, the 'none' key must be mutually exclusive. Add M2-specific logic to the chip tap handler:

```typescript
function handleChipTap(key: string) {
  setChipSelections((prev) => {
    // M2 safety wall: 'none' is mutually exclusive with all allergen chips.
    // Tapping 'none' clears everything and selects only 'none'.
    // Tapping any allergen clears 'none' and toggles the allergen.
    if (currentMomentKey === 'm2_safe') {
      if (key === 'none') {
        return prev.includes('none') ? [] : ['none'];
      }
      const withoutNone = prev.filter((k) => k !== 'none');
      return withoutNone.includes(key)
        ? withoutNone.filter((k) => k !== key)
        : [...withoutNone, key];
    }
    // Default multi-select toggle (M3–M5 moments that use choice mode)
    return prev.includes(key) ? prev.filter((k) => k !== key) : [...prev, key];
  });
}
```

The chip tap handler is currently part of the inline JSX chip rendering. Extract it into a named `handleChipTap` callback (if not already extracted). Pass `currentMomentKey` to it via closure (it's already in scope).

## Dev Notes

### Encryption pattern for `child_allergens`

The `ChildAllergensRepository` follows the same pattern as `HouseholdsRepository`:
1. Get DEK via `getOrCreateHouseholdDek(this.client, this.kek, householdId)`
2. Encrypt allergen string with `encryptField(allergen, dek)` — stores AES-256-GCM ciphertext
3. Compute hash with `normalizedHash(allergen)` — SHA-256(lower(trim(plaintext)))
4. Upsert with `(child_id, allergen_hash)` as the conflict target

The `kek` (Key Encryption Key) comes from `fastify.env.ENVELOPE_ENCRYPTION_MASTER_KEY` — same as `HouseholdsRepository`. In `onboarding.routes.ts`, it's already on `kek` local variable (constructed on line 43 of the current routes file).

### "No known allergens" — what happens server-side

When the parent selects the 'none' chip, the chip serializer sends `[Chips selected: no known allergens]` to the agent. The v2 agent prompt instructs the agent to **not call allergen.declare** in this case — instead it acknowledges the response and emits `[NEXT_MOMENT:m3_taste]`. The service then:
1. `advancedOutOfM2 = true` (because nextCurrentMoment !== 'm2_safe' AND previousMoment === 'm2_safe')
2. `m2_allergen_response = advancedOutOfM2` → `true`
3. Required-set bit latches; conversation can proceed to finalize gate

This is already implemented in `onboarding.service.ts` — no server-side change needed.

### Chip tap handler location in `OnboardingText.tsx`

The current chip rendering in `OnboardingText.tsx` renders `ChoiceChip` components inside the focused-view section. The click handler passes a `key` to toggle `chipSelections`. Find the section with `chipConfig?.mode !== 'hint'` (or the `ChoiceChip` rendering block) and add the M2-specific mutual exclusion logic to the handler.

The handler must be a `useCallback` that reads `currentMomentKey` from closure (it's already in scope as state). No prop drilling needed.

### `dietary_preferences` upsert functional index workaround

Supabase `.upsert()` with `onConflict: 'household_id,child_id,tag'` may fail because the unique index is a functional expression (`COALESCE(child_id, sentinel_uuid)`). If this fails at runtime with a 42P10 error (no unique or exclusion constraint matching ON CONFLICT):

```typescript
// Alternative: insert, catch 23505, select existing
try {
  const { data, error } = await this.client
    .from('dietary_preferences')
    .insert({ household_id, child_id, tag, enforcement, source })
    .select('id')
    .single();
  if (error) throw error;
  return { dietary_id: data.id, was_existing: false };
} catch (err: unknown) {
  if (isUniqueViolation(err)) {
    const sentinelUuid = '00000000-0000-0000-0000-000000000000';
    const { data } = await this.client
      .from('dietary_preferences')
      .select('id')
      .eq('household_id', householdId)
      .eq(child_id === null ? 'child_id' : 'child_id', child_id === null ? sentinelUuid : child_id)
      .eq('tag', tag)
      .maybeSingle();
    return { dietary_id: data!.id, was_existing: true };
  }
  throw err;
}
```

Import `isUniqueViolation` from `../../common/errors.js`.

### `validatedDietary` in `dietary.declare` tool

The `vocabularyService.validateDietary([parsed.tag])` call will throw if the tag is not in the active dietary vocabulary. This is intentional — the agent receives the error as a tool-call result and can recover by choosing a valid tag from the vocabulary block in the system prompt. Do NOT swallow this error.

The 5 failing `createChildUpsertToolSpec` tests have a pre-existing `resolveAllergen` API drift (tracked but not fixed in prior slices). Do NOT fix or modify these tests in this slice — they must remain as-is. The fix is blocked on a vocabulary service refactor that's out of scope.
